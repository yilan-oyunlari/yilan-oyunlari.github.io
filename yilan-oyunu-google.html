// --- Advanced Game Logic --- let score = 0; let highScore = localStorage.getItem("snakeZillaHighScore") || 0; let level = "EASY"; let speed = 150; // Initial speed let foodCount = 0; function update() { let head = {x: snake[0].x + dx, y: snake[0].y + dy}; // Wall Collision Check if(head.x < 0 || head.x >= tileCount || head.y < 0 || head.y >= tileCount || snake.some(s => s.x === head.x && s.y === head.y)) { gameOver(); return; } snake.unshift(head); // Food logic if(head.x === food.x && head.y === food.y) { // Super Food Score logic let points = (food.type === 'super') ? 10 : 1; score += points; foodCount++; // Dynamic Difficulty (Level up logic) updateDifficulty(); document.getElementById("scoreVal").innerText = score; spawnFood(); } else { snake.pop(); } draw(); } function updateDifficulty() { // Speed badhane ka logic if (score > 10) { speed = 120; level = "MEDIUM"; } if (score > 30) { speed = 80; level = "HARD"; } document.getElementById("levelVal").innerText = level; } function spawnFood() { // 5 points ke baad Super Food spawn hoga let isSuper = (foodCount > 0 && foodCount % 5 === 0); food = { x: Math.floor(Math.random() * tileCount), y: Math.floor(Math.random() * tileCount), type: isSuper ? 'super' : 'normal' }; } // Drawing logic mein ye update karein takki Super Food bada dikhe function drawFood() { if(food.type === 'super') { ctx.fillStyle = "#FFD700"; // Gold color ctx.shadowBlur = 20; ctx.shadowColor = "#FFD700"; ctx.fillRect(food.x * gridSize + 2, food.y * gridSize + 2, gridSize - 4, gridSize - 4); } else { ctx.fillStyle = "#ff003c"; ctx.fillRect(food.x * gridSize + 4, food.y * gridSize + 4, gridSize - 8, gridSize - 8); } ctx.shadowBlur = 0; }
